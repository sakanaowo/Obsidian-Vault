8.4.7 — Thêm renderCalculatorMode()

Thêm method:

/**
 * Hiển thị đúng layout tương ứng currentMode.
 *
 * Chỉ một mode được VISIBLE.
 */
private void renderCalculatorMode() {
    if (currentMode == null) {
        currentMode =
                CalculatorMode.BASIC;
    }
    boolean showBasic =
            currentMode
                    == CalculatorMode.BASIC;
    boolean showScientific =
            currentMode
                    == CalculatorMode.SCIENTIFIC;
    boolean showConvert =
            currentMode
                    == CalculatorMode.CONVERT;
    basicCalculatorLayout.setVisibility(
            showBasic
                    ? View.VISIBLE
                    : View.GONE
    );
    scientificCalculatorLayout.setVisibility(
            showScientific
                    ? View.VISIBLE
                    : View.GONE
    );
    converterLayout.setVisibility(
            showConvert
                    ? View.VISIBLE
                    : View.GONE
    );
}

Logic:

BASIC
Basic       VISIBLE
Scientific  GONE
Convert     GONE
SCIENTIFIC
Basic       GONE
Scientific  VISIBLE
Convert     GONE
CONVERT
Basic       GONE
Scientific  GONE
Convert     VISIBLE

private void setCalculatorMode(

        return;
    }
    /*
     * Update state.

    currentMode =

    renderCalculatorMode();
