7.9.2 — Cập nhật HistoryBottomSheet.java

File:

app/src/main/java/com/example/calculator/ui/HistoryBottomSheet.java

Có thể thay toàn bộ bằng phiên bản dưới đây để tránh bỏ sót một method.

package com.example.calculator.ui;
import android.os.Bundle;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.TextView;
import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import androidx.appcompat.app.AlertDialog;
import androidx.recyclerview.widget.LinearLayoutManager;
import androidx.recyclerview.widget.RecyclerView;
import com.example.calculator.R;
import com.example.calculator.history.HistoryManager;
import com.example.calculator.model.HistoryItem;
import com.example.calculator.ui.adapter.HistoryAdapter;
import com.google.android.material.bottomsheet.BottomSheetDialogFragment;
import java.util.List;
/**
 * Bottom Sheet hiển thị và quản lý Calculation History.
 *
 * Trách nhiệm:
 *
 * - Load history từ local storage
 * - Hiển thị RecyclerView
 * - Empty state
 * - Delete một calculation
 * - Clear toàn bộ history
 * - Close sheet
 */
public class HistoryBottomSheet
        extends BottomSheetDialogFragment {
    // =========================================================
    // DATA
    // =========================================================
    private HistoryManager historyManager;
    private HistoryAdapter historyAdapter;
    // =========================================================
    // VIEW
    // =========================================================
    private RecyclerView recyclerHistory;
    private TextView tvEmptyHistory;
    private TextView btnClearHistory;
    // =========================================================
    // CREATE VIEW
    // =========================================================
    @Nullable
    @Override
    public View onCreateView(
            @NonNull LayoutInflater inflater,
            @Nullable ViewGroup container,
            @Nullable Bundle savedInstanceState
    ) {
        return inflater.inflate(
                R.layout.bottom_sheet_history,
                container,
                false
        );
    }
    // =========================================================
    // VIEW CREATED
    // =========================================================
    @Override
    public void onViewCreated(
            @NonNull View view,
            @Nullable Bundle savedInstanceState
    ) {
        super.onViewCreated(
                view,
                savedInstanceState
        );
        historyManager =
                new HistoryManager(
                        requireContext()
                                .getApplicationContext()
                );
        bindViews(view);
        setupRecyclerView();
        setupCloseButton(view);
        setupClearButton();
        loadHistory();
    }
    // =========================================================
    // BIND VIEWS
    // =========================================================
    private void bindViews(
            @NonNull View rootView
    ) {
        recyclerHistory =
                rootView.findViewById(
                        R.id.recyclerHistory
                );
        tvEmptyHistory =
                rootView.findViewById(
                        R.id.tvEmptyHistory
                );
        btnClearHistory =
                rootView.findViewById(
                        R.id.btnClearHistory
                );
    }
    // =========================================================
    // RECYCLER VIEW
    // =========================================================
    private void setupRecyclerView() {
        recyclerHistory.setLayoutManager(
                new LinearLayoutManager(
                        requireContext()
                )
        );
        /*
         * Callback long press:
         *
         * Adapter
         *    ↓
         * HistoryBottomSheet
         *    ↓
         * confirmation dialog
         *    ↓
         * HistoryManager.delete()
         */
        historyAdapter =
                new HistoryAdapter(
                        null,
                        this::showDeleteConfirmation
                );
        recyclerHistory.setAdapter(
                historyAdapter
        );
    }
    // =========================================================
    // CLOSE
    // =========================================================
    private void setupCloseButton(
            @NonNull View rootView
    ) {
        rootView
                .findViewById(
                        R.id.btnCloseHistory
                )
                .setOnClickListener(view -> {
                    dismiss();
                });
    }
    // =========================================================
    // CLEAR BUTTON
    // =========================================================
    private void setupClearButton() {
        btnClearHistory.setOnClickListener(view -> {
            showClearHistoryConfirmation();
        });
    }
    // =========================================================
    // LOAD HISTORY
    // =========================================================
    /**
     * Reload toàn bộ dữ liệu từ local storage.
     *
     * Method này được gọi khi:
     *
     * - sheet mở
     * - delete một item
     * - clear all
     */
    private void loadHistory() {
        List<HistoryItem> history =
                historyManager.getAll();
        historyAdapter.submitList(
                history
        );
        renderHistoryState(
                history
        );
    }
    // =========================================================
    // DELETE ONE
    // =========================================================
    /**
     * Hiển thị confirmation trước khi xóa một calculation.
     *
     * Được gọi từ callback long-press của HistoryAdapter.
     */
    private void showDeleteConfirmation(
            HistoryItem item
    ) {
        if (item == null) {
            return;
        }
        new AlertDialog.Builder(
                requireContext()
        )
                .setTitle(
                        R.string.delete_history_title
                )
                .setMessage(
                        R.string.delete_history_message
                )
                .setNegativeButton(
                        R.string.action_cancel,
                        null
                )
                .setPositiveButton(
                        R.string.action_delete,
                        (dialog, which) -> {
                            deleteHistoryItem(
                                    item
                            );
                        }
                )
                .show();
    }
    /**
     * Xóa item khỏi SharedPreferences
     * rồi reload RecyclerView.
     */
    private void deleteHistoryItem(
            HistoryItem item
    ) {
        historyManager.delete(
                item.getId()
        );
        /*
         * Không tự sửa list trong Adapter.
         *
         * Storage là source of truth:
         *
         * delete
         * ↓
         * getAll
         * ↓
         * submitList
         */
        loadHistory();
    }
    // =========================================================
    // CLEAR ALL
    // =========================================================
    /**
     * Confirmation trước khi xóa toàn bộ History.
     */
    private void showClearHistoryConfirmation() {
        /*
         * Không mở dialog nếu history đã rỗng.
         */
        if (historyManager.size() == 0) {
            return;
        }
        new AlertDialog.Builder(
                requireContext()
        )
                .setTitle(
                        R.string.clear_history_title
                )
                .setMessage(
                        R.string.clear_history_message
                )
                .setNegativeButton(
                        R.string.action_cancel,
                        null
                )
                .setPositiveButton(
                        R.string.action_clear,
                        (dialog, which) -> {
                            clearAllHistory();
                        }
                )
                .show();
    }
    /**
     * Xóa toàn bộ dữ liệu History.
     */
    private void clearAllHistory() {
        historyManager.clear();
        /*
         * Sau clear:
         *
         * RecyclerView → hidden
         * Empty state  → visible
         * Clear button → hidden
         */
        loadHistory();
    }
    // =========================================================
    // STATE
    // =========================================================
    private void renderHistoryState(
            List<HistoryItem> history
    ) {
        boolean isEmpty =
                history == null
                        || history.isEmpty();
        if (isEmpty) {
            recyclerHistory.setVisibility(
                    View.GONE
            );
            tvEmptyHistory.setVisibility(
                    View.VISIBLE
            );
            btnClearHistory.setVisibility(
                    View.GONE
            );
        } else {
            recyclerHistory.setVisibility(
                    View.VISIBLE
            );
            tvEmptyHistory.setVisibility(
                    View.GONE
            );
            btnClearHistory.setVisibility(
                    View.VISIBLE
            );
        }
    }
}

7.9.5 — Vì sao reload từ storage thay vì xóa trực tiếp Adapter?

Không làm:

items.remove(position);
notifyItemRemoved(position);

trong HistoryBottomSheet.

Ta đang giữ:

SharedPreferences
      =
Source of Truth

Flow:

HistoryManager.delete()
        ↓
SharedPreferences cập nhật
        ↓
HistoryManager.getAll()
        ↓
HistoryAdapter.submitList()

Điều này tránh trường hợp:

UI đã xóa
nhưng storage chưa xóa

hoặc ngược lại.

Với tối đa khoảng 100 item, reload toàn bộ list là đủ nhẹ cho MVP.
