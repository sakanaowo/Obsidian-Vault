package com.example.calculator.model;
import java.util.Objects;
/**
 * Mô tả một đơn vị đo offline.
 *
 * Ví dụ:
 *
 * Length:
 *
 * Meter
 * symbol       = "m"
 * factorToBase = 1.0
 *
 * Kilometer
 * symbol       = "km"
 * factorToBase = 1000.0
 *
 * Centimeter
 * symbol       = "cm"
 * factorToBase = 0.01
 *
 * Với các category thông thường:
 *
 * input
 *   ↓
 * input × factorToBase
 *   ↓
 * base unit
 *
 * Temperature sẽ được xử lý riêng trong
 * UnitConverterEngine vì cần cả scale và offset.
 */
public final class UnitDefinition {
    // =========================================================
    // DATA
    // =========================================================
    private final UnitCategory category;
    private final String name;
    private final String symbol;
    private final double factorToBase;
    // =========================================================
    // CONSTRUCTOR
    // =========================================================
    public UnitDefinition(
            UnitCategory category,
            String name,
            String symbol,
            double factorToBase
    ) {
        if (category == null) {
            throw new IllegalArgumentException(
                    "Unit category cannot be null."
            );
        }
        if (name == null
                || name.trim().isEmpty()) {
            throw new IllegalArgumentException(
                    "Unit name cannot be empty."
            );
        }
        if (symbol == null
                || symbol.trim().isEmpty()) {
            throw new IllegalArgumentException(
                    "Unit symbol cannot be empty."
            );
        }
        if (Double.isNaN(factorToBase)
                || Double.isInfinite(factorToBase)
                || factorToBase <= 0.0) {
            throw new IllegalArgumentException(
                    "factorToBase must be a positive finite number."
            );
        }
        this.category =
                category;
        this.name =
                name.trim();
        this.symbol =
                symbol.trim();
        this.factorToBase =
                factorToBase;
    }
    // =========================================================
    // GETTERS
    // =========================================================
    public UnitCategory getCategory() {
        return category;
    }
    public String getName() {
        return name;
    }
    public String getSymbol() {
        return symbol;
    }
    public double getFactorToBase() {
        return factorToBase;
    }
    // =========================================================
    // CATEGORY CHECK
    // =========================================================
    public boolean belongsTo(
            UnitCategory expectedCategory
    ) {
        return category == expectedCategory;
    }
    // =========================================================
    // OBJECT CONTRACT
    // =========================================================
    @Override
    public boolean equals(
            Object object
    ) {
        if (this == object) {
            return true;
        }
        if (!(object instanceof UnitDefinition)) {
            return false;
        }
        UnitDefinition other =
                (UnitDefinition) object;
        return category == other.category
                && name.equals(
                        other.name
                )
                && symbol.equals(
                        other.symbol
                );
    }
    @Override
    public int hashCode() {
        return Objects.hash(
                category,
                name,
                symbol
        );
    }
    @Override
    public String toString() {
        return name
                + " ("
                + symbol
                + ")";
    }
}

Centimeter

Ví dụ đổi:

1 km → meter

Bước về base:

1 × 1000
= 1000 m

Đổi:

100 cm → meter

Bước về base:

100 × 0.01
= 1 m

input

→ target unit
