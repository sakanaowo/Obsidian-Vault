package com.example.calculator.ui.adapter;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.TextView;
import androidx.annotation.NonNull;
import androidx.annotation.StringRes;
import androidx.recyclerview.widget.RecyclerView;
import com.example.calculator.R;
import com.example.calculator.model.HistoryItem;
import java.util.ArrayList;
import java.util.List;
/**
 * Adapter hiển thị Calculation History có grouping theo timestamp.
 *
 * Ví dụ:
 *
 * Previous 7 Days
 *     7 × 8
 *     56
 *
 *     2 + 3
 *     5
 *
 * Previous 30 Days
 *     ...
 */
public class HistoryAdapter
        extends RecyclerView.Adapter<RecyclerView.ViewHolder> {
    // =========================================================
    // VIEW TYPES
    // =========================================================
    private static final int VIEW_TYPE_SECTION = 0;
    private static final int VIEW_TYPE_HISTORY = 1;
    // =========================================================
    // TIME
    // =========================================================
    private static final long DAY_IN_MILLIS =
            24L * 60L * 60L * 1000L;
    private static final long SEVEN_DAYS_IN_MILLIS =
            7L * DAY_IN_MILLIS;
    private static final long THIRTY_DAYS_IN_MILLIS =
            30L * DAY_IN_MILLIS;
    // =========================================================
    // DATA
    // =========================================================
    /*
     * Không render trực tiếp List<HistoryItem>.
     *
     * Ta chuyển thành:
     *
     * SECTION
     * HISTORY
     * HISTORY
     * SECTION
     * HISTORY
     */
    private final List<HistoryRow> rows =
            new ArrayList<>();
    private final OnHistoryItemLongClickListener
            longClickListener;
    // =========================================================
    // CALLBACK
    // =========================================================
    public interface OnHistoryItemLongClickListener {
        void onHistoryItemLongClick(
                HistoryItem item
        );
    }
    // =========================================================
    // CONSTRUCTOR
    // =========================================================
    public HistoryAdapter(
            List<HistoryItem> initialItems,
            OnHistoryItemLongClickListener longClickListener
    ) {
        this.longClickListener =
                longClickListener;
        if (initialItems != null) {
            buildRows(
                    initialItems
            );
        }
    }
    // =========================================================
    // VIEW TYPE
    // =========================================================
    @Override
    public int getItemViewType(
            int position
    ) {
        HistoryRow row =
                rows.get(position);
        return row.isSection()
                ? VIEW_TYPE_SECTION
                : VIEW_TYPE_HISTORY;
    }
    // =========================================================
    // CREATE
    // =========================================================
    @NonNull
    @Override
    public RecyclerView.ViewHolder onCreateViewHolder(
            @NonNull ViewGroup parent,
            int viewType
    ) {
        LayoutInflater inflater =
                LayoutInflater.from(
                        parent.getContext()
                );
        if (viewType == VIEW_TYPE_SECTION) {
            View view =
                    inflater.inflate(
                            R.layout.item_history_section,
                            parent,
                            false
                    );
            return new SectionViewHolder(
                    view
            );
        }
        View view =
                inflater.inflate(
                        R.layout.item_history,
                        parent,
                        false
                );
        return new HistoryViewHolder(
                view
        );
    }
    // =========================================================
    // BIND
    // =========================================================
    @Override
    public void onBindViewHolder(
            @NonNull RecyclerView.ViewHolder holder,
            int position
    ) {
        HistoryRow row =
                rows.get(position);
        if (holder instanceof SectionViewHolder) {
            ((SectionViewHolder) holder)
                    .bind(
                            row.getSectionTitleRes()
                    );
            return;
        }
        if (holder instanceof HistoryViewHolder) {
            ((HistoryViewHolder) holder)
                    .bind(
                            row.getHistoryItem(),
                            longClickListener
                    );
        }
    }
    // =========================================================
    // COUNT
    // =========================================================
    @Override
    public int getItemCount() {
        return rows.size();
    }
    // =========================================================
    // UPDATE DATA
    // =========================================================
    /**
     * Nhận raw HistoryItem list,
     * sau đó rebuild section + item rows.
     */
    public void submitList(
            List<HistoryItem> newItems
    ) {
        rows.clear();
        if (newItems != null) {
            buildRows(
                    newItems
            );
        }
        notifyDataSetChanged();
    }
    // =========================================================
    // BUILD GROUPS
    // =========================================================
    private void buildRows(
            List<HistoryItem> history
    ) {
        if (history == null
                || history.isEmpty()) {
            return;
        }
        int previousSection = -1;
        for (HistoryItem item : history) {
            int currentSection =
                    getSectionForTimestamp(
                            item.getTimestamp()
                    );
            /*
             * Chỉ thêm section header khi section thay đổi.
             *
             * Vì HistoryManager đã giữ:
             *
             * newest
             * ↓
             * oldest
             *
             * nên các item cùng thời gian sẽ nằm cạnh nhau.
             */
            if (currentSection
                    != previousSection) {
                rows.add(
                        HistoryRow.section(
                                currentSection
                        )
                );
                previousSection =
                        currentSection;
            }
            rows.add(
                    HistoryRow.history(
                            item
                    )
            );
        }
    }
    /**
     * Timestamp
     *     ↓
     * section string resource
     */
    @StringRes
    private int getSectionForTimestamp(
            long timestamp
    ) {
        long now =
                System.currentTimeMillis();
        long age =
                now - timestamp;
        /*
         * Nếu clock thay đổi làm timestamp nằm trong tương lai,
         * coi item đó là recent thay vì crash hoặc group sai.
         */
        if (age < 0) {
            age = 0;
        }
        if (age <= SEVEN_DAYS_IN_MILLIS) {
            return R.string.history_previous_7_days;
        }
        if (age <= THIRTY_DAYS_IN_MILLIS) {
            return R.string.history_previous_30_days;
        }
        return R.string.history_older;
    }
    // =========================================================
    // SECTION VIEW HOLDER
    // =========================================================
    static class SectionViewHolder
            extends RecyclerView.ViewHolder {
        private final TextView tvSection;
        public SectionViewHolder(
                @NonNull View itemView
        ) {
            super(itemView);
            tvSection =
                    itemView.findViewById(
                            R.id.tvHistorySection
                    );
        }
        public void bind(
                @StringRes int titleRes
        ) {
            tvSection.setText(
                    titleRes
            );
        }
    }
    // =========================================================
    // HISTORY VIEW HOLDER
    // =========================================================
    static class HistoryViewHolder
            extends RecyclerView.ViewHolder {
        private final TextView tvExpression;
        private final TextView tvResult;
        public HistoryViewHolder(
                @NonNull View itemView
        ) {
            super(itemView);
            tvExpression =
                    itemView.findViewById(
                            R.id.tvHistoryExpression
                    );
            tvResult =
                    itemView.findViewById(
                            R.id.tvHistoryResult
                    );
        }
        public void bind(
                HistoryItem item,
                OnHistoryItemLongClickListener longClickListener
        ) {
            if (item == null) {
                return;
            }
            tvExpression.setText(
                    item.getExpression()
            );
            tvResult.setText(
                    item.getResult()
            );
            /*
             * Long press chỉ áp dụng cho calculation row.
             *
             * Section header không có interaction delete.
             */
            if (longClickListener != null) {
                itemView.setOnLongClickListener(view -> {
                    longClickListener
                            .onHistoryItemLongClick(
                                    item
                            );
                    return true;
                });
            } else {
                itemView.setOnLongClickListener(
                        null
                );
            }
        }
    }
    // =========================================================
    // INTERNAL ROW MODEL
    // =========================================================
    /**
     * Internal model của Adapter.
     *
     * Một row chỉ có một trong hai:
     *
     * sectionTitleRes
     *
     * hoặc
     *
     * historyItem
     */
    private static class HistoryRow {
        @StringRes
        private final int sectionTitleRes;
        private final HistoryItem historyItem;
        private HistoryRow(
                @StringRes int sectionTitleRes,
                HistoryItem historyItem
        ) {
            this.sectionTitleRes =
                    sectionTitleRes;
            this.historyItem =
                    historyItem;
        }
        static HistoryRow section(
                @StringRes int titleRes
        ) {
            return new HistoryRow(
                    titleRes,
                    null
            );
        }
        static HistoryRow history(
                HistoryItem item
        ) {
            return new HistoryRow(
                    0,
                    item
            );
        }
        boolean isSection() {
            return historyItem == null;
        }
        @StringRes
        int getSectionTitleRes() {
            return sectionTitleRes;
        }
        HistoryItem getHistoryItem() {
            return historyItem;
        }
    }
}
