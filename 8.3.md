8.3.6 — Thêm showModePopup()

Thêm method:

/**
 * Inflate popup_calculator_mode.xml
 * và hiển thị bên dưới btnMode.
 */
private void showModePopup(
        View anchor
) {
    /*
     * Inflate layout đã tạo ở bước 8.2.
     */
    View popupView =
            LayoutInflater
                    .from(this)
                    .inflate(
                            R.layout.popup_calculator_mode,
                            null,
                            false
                    );
    /*
     * Tạo PopupWindow.
     *
     * Root XML đã có width = 280dp,
     * nên PopupWindow có thể dùng WRAP_CONTENT.
     */
    modePopupWindow =
            new PopupWindow(
                    popupView,
                    ViewGroup.LayoutParams.WRAP_CONTENT,
                    ViewGroup.LayoutParams.WRAP_CONTENT,
                    true
            );
    /*
     * Background transparent ở level PopupWindow.
     *
     * Background thật nằm ở:
     *
     * bg_mode_popup.xml
     */
    modePopupWindow.setBackgroundDrawable(
            new ColorDrawable(
                    Color.TRANSPARENT
            )
    );
    /*
     * Cho phép tap ngoài popup để đóng.
     */
    modePopupWindow.setOutsideTouchable(
            true
    );
    /*
     * Shadow nhẹ.
     */
    modePopupWindow.setElevation(
            getResources()
                    .getDimension(
                            R.dimen.spacing_8
                    )
    );
    /*
     * Đồng bộ checkmark với currentMode
     * TRƯỚC khi popup hiển thị.
     */
    updateModeCheckmarks(
            popupView
    );
    /*
     * Setup click cho 3 mode.
     */
    setupModePopupActions(
            popupView
    );
    /*
     * Khi popup đóng,
     * xóa reference cũ.
     */
    modePopupWindow.setOnDismissListener(
            () -> modePopupWindow = null
    );
    /*
     * Measure để popup căn phải theo btnMode.
     *
     * Nếu chỉ dùng:
     *
     * showAsDropDown(anchor)
     *
     * popup 280dp sẽ bắt đầu từ mép trái của btnMode
     * và có thể tràn khỏi cạnh phải màn hình.
     */
    popupView.measure(
            View.MeasureSpec.makeMeasureSpec(
                    0,
                    View.MeasureSpec.UNSPECIFIED
            ),
            View.MeasureSpec.makeMeasureSpec(
                    0,
                    View.MeasureSpec.UNSPECIFIED
            )
    );
    int popupWidth =
            popupView.getMeasuredWidth();
    /*
     * Căn cạnh phải popup
     * với cạnh phải btnMode.
     */
    int xOffset =
            anchor.getWidth()
                    - popupWidth;
    /*
     * Khoảng cách nhỏ giữa btnMode và popup.
     */
    int yOffset =
            getResources()
                    .getDimensionPixelSize(
                            R.dimen.spacing_8
                    );
    modePopupWindow.showAsDropDown(
            anchor,
            xOffset,
            yOffset
    );
}
