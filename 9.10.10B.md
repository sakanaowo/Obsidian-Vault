package com.example.calculator.ui.adapter;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.ImageView;
import android.widget.TextView;
import androidx.annotation.NonNull;
import androidx.recyclerview.widget.RecyclerView;
import com.example.calculator.R;
import com.example.calculator.model.UnitDefinition;
import java.util.ArrayList;
import java.util.List;
/**
 * Adapter hiển thị danh sách UnitDefinition.
 *
 * Trách nhiệm:
 *
 * - Inflate item_unit.xml
 * - Hiển thị name và symbol
 * - Quản lý unit đang được chọn
 * - Forward selection event
 *
 * Adapter không thực hiện conversion.
 */
public final class UnitAdapter
        extends RecyclerView.Adapter<UnitAdapter.UnitViewHolder> {
    // =========================================================
    // CALLBACK
    // =========================================================
    public interface OnUnitSelectedListener {
        void onUnitSelected(
                UnitDefinition unit
        );
    }
    // =========================================================
    // DATA
    // =========================================================
    private final List<UnitDefinition> units =
            new ArrayList<>();
    private final OnUnitSelectedListener
            unitSelectedListener;
    private UnitDefinition selectedUnit;
    // =========================================================
    // CONSTRUCTOR
    // =========================================================
    public UnitAdapter(
            OnUnitSelectedListener unitSelectedListener
    ) {
        this.unitSelectedListener =
                unitSelectedListener;
    }
    // =========================================================
    // CREATE VIEW HOLDER
    // =========================================================
    @NonNull
    @Override
    public UnitViewHolder onCreateViewHolder(
            @NonNull ViewGroup parent,
            int viewType
    ) {
        View itemView =
                LayoutInflater
                        .from(parent.getContext())
                        .inflate(
                                R.layout.item_unit,
                                parent,
                                false
                        );
        return new UnitViewHolder(
                itemView
        );
    }
    // =========================================================
    // BIND
    // =========================================================
    @Override
    public void onBindViewHolder(
            @NonNull UnitViewHolder holder,
            int position
    ) {
        UnitDefinition unit =
                units.get(position);
        boolean selected =
                selectedUnit != null
                        && selectedUnit.equals(
                                unit
                        );
        holder.bind(
                unit,
                selected,
                this::selectUnit
        );
    }
    // =========================================================
    // COUNT
    // =========================================================
    @Override
    public int getItemCount() {
        return units.size();
    }
    // =========================================================
    // PUBLIC DATA API
    // =========================================================
    /**
     * Thay danh sách đang hiển thị.
     *
     * Dùng khi:
     *
     * - mở picker;
     * - search query thay đổi.
     */
    public void submitList(
            List<UnitDefinition> newUnits
    ) {
        units.clear();
        if (newUnits != null) {
            units.addAll(
                    newUnits
            );
        }
        notifyDataSetChanged();
    }
    /**
     * Đặt unit được chọn ban đầu.
     */
    public void setSelectedUnit(
            UnitDefinition unit
    ) {
        selectedUnit =
                unit;
        notifyDataSetChanged();
    }
    public UnitDefinition getSelectedUnit() {
        return selectedUnit;
    }
    // =========================================================
    // SELECTION
    // =========================================================
    private void selectUnit(
            UnitDefinition unit
    ) {
        if (unit == null) {
            return;
        }
        if (unit.equals(
                selectedUnit
        )) {
            return;
        }
        selectedUnit =
                unit;
        notifyDataSetChanged();
        if (unitSelectedListener != null) {
            unitSelectedListener
                    .onUnitSelected(
                            unit
                    );
        }
    }
    // =========================================================
    // VIEW HOLDER
    // =========================================================
    static final class UnitViewHolder
            extends RecyclerView.ViewHolder {
        private final TextView tvUnitName;
        private final TextView tvUnitSymbol;
        private final ImageView ivUnitSelected;
        UnitViewHolder(
                @NonNull View itemView
        ) {
            super(itemView);
            tvUnitName =
                    itemView.findViewById(
                            R.id.tvUnitName
                    );
            tvUnitSymbol =
                    itemView.findViewById(
                            R.id.tvUnitSymbol
                    );
            ivUnitSelected =
                    itemView.findViewById(
                            R.id.ivUnitSelected
                    );
        }
        void bind(
                UnitDefinition unit,
                boolean selected,
                OnUnitSelectedListener listener
        ) {
            tvUnitName.setText(
                    unit.getName()
            );
            tvUnitSymbol.setText(
                    unit.getSymbol()
            );
            ivUnitSelected.setVisibility(
                    selected
                            ? View.VISIBLE
                            : View.GONE
            );
            itemView.setOnClickListener(view -> {
                if (listener != null) {
                    listener.onUnitSelected(
                            unit
                    );
                }
            });
        }
    }
}

    private String target;
    private String currentUnitSymbol;
    /**
     * Danh sách gốc của category hiện tại.
     *
     * Search chỉ lọc từ list này.
     */

            new ArrayList<>();
    /**
     * Unit đang được chọn tạm thời.
     *
     * Converter chỉ nhận unit này khi user nhấn Confirm.
     */
    private UnitDefinition pendingSelectedUnit;
    // =========================================================
    // ADAPTER
    // =========================================================
    private UnitAdapter unitAdapter;
    // =========================================================
    // VIEWS
    // =========================================================
    private TextView tvCategory;
    private TextView tvEmptyUnits;
    private EditText etSearch;
    private RecyclerView recyclerUnits;
    private Button btnConfirm;
    // =========================================================
    // FACTORY
    // =========================================================
    /**
     * Tạo Unit Picker.
     *
     * @param category    category hiện tại
     * @param currentUnit unit đang được chọn
     * @param target      TARGET_FROM hoặc TARGET_TO
     */
    public static UnitPickerBottomSheet newInstance(
            UnitCategory category,
            UnitDefinition currentUnit,
            String target
    ) {
        UnitPickerBottomSheet bottomSheet =
                new UnitPickerBottomSheet();
        Bundle arguments =
                new Bundle();
        if (category != null) {
            arguments.putString(
                    ARG_CATEGORY,
                    category.name()
            );
        }
        if (currentUnit != null) {
            arguments.putString(
                    ARG_CURRENT_UNIT_SYMBOL,
                    currentUnit.getSymbol()
            );
        }
        arguments.putString(
                ARG_TARGET,
                normalizeTarget(target)
        );
        bottomSheet.setArguments(
                arguments
        );
        return bottomSheet;
    }
    private static String normalizeTarget(
            String target
    ) {
        if (TARGET_TO.equals(target)) {
            return TARGET_TO;
        }
        return TARGET_FROM;
    }

    // FRAGMENT LIFECYCLE
    // =========================================================
    @Override
    public void onCreate(
            @Nullable Bundle savedInstanceState
    ) {
        super.onCreate(
                savedInstanceState
        );
        readArguments();
    }
    @Nullable
    @Override
    public View onCreateView(
            @NonNull LayoutInflater inflater,

                bottomSheetDialog.findViewById(
                        com.google.android.material.R.id.design_bottom_sheet
                );
        if (sheet == null) {
            return;
        }

                sheet.getLayoutParams();
        layoutParams.height =
                ViewGroup.LayoutParams.MATCH_PARENT;

                layoutParams
        );

                BottomSheetBehavior.from(
                        sheet
                );
        behavior.setSkipCollapsed(
                true
        );
        behavior.setState(

        );
    }
    @Override
    public void onDestroyView() {
        super.onDestroyView();

                null;

            target =
                    TARGET_FROM;
            currentUnitSymbol =
                    null;
            return;
        }
        String categoryName =
                arguments.getString(
                        ARG_CATEGORY
                );
        category =
                parseCategory(
                        categoryName
                );
        currentUnitSymbol =
                arguments.getString(
                        ARG_CURRENT_UNIT_SYMBOL
                );
        target =
                normalizeTarget(
                        arguments.getString(
                                ARG_TARGET
                        )
                );
    }
    private UnitCategory parseCategory(
            String categoryName
    ) {
        if (categoryName == null
                || categoryName.trim().isEmpty()) {
            return null;
        }
        try {
            return UnitCategory.valueOf(
                    categoryName
            );
        } catch (IllegalArgumentException exception) {
            return null;
        }
    }
    // =========================================================
    // BIND VIEWS
    // =========================================================
    private void bindViews(
            @NonNull View rootView
    ) {
        tvCategory =
                requireView(
                        rootView,
                        R.id.tvUnitPickerCategory
                );
        tvEmptyUnits =
                requireView(
                        rootView,
                        R.id.tvEmptyUnits
                );
        etSearch =
                requireView(
                        rootView,
                        R.id.etUnitSearch
                );
        recyclerUnits =
                requireView(
                        rootView,
                        R.id.recyclerUnits
                );
        btnConfirm =
                requireView(
                        rootView,
                        R.id.btnConfirmUnit
                );
    }
    private <T extends View> T requireView(
            View rootView,
            int viewId
    ) {
        T view =
                rootView.findViewById(
                        viewId
                );
        if (view == null) {
            throw new IllegalStateException(
                    "Missing Unit Picker View ID: "
                            + viewId
            );
        }
        return view;
    }
    // =========================================================
    // RECYCLER VIEW
    // =========================================================
    private void setupRecyclerView() {
        recyclerUnits.setLayoutManager(
                new LinearLayoutManager(
                        requireContext()
                )
        );
        unitAdapter =
                new UnitAdapter(
                        this::onTemporaryUnitSelected
                );
        recyclerUnits.setAdapter(
                unitAdapter
        );
    }
    /**
     * Chỉ thay lựa chọn tạm thời.
     *
     * Converter chưa được cập nhật tại đây.
     */
    private void onTemporaryUnitSelected(
            UnitDefinition unit
    ) {
        if (unit == null) {
            return;
        }
        pendingSelectedUnit =
                unit;
        btnConfirm.setEnabled(
                true
        );
    }
    // =========================================================
    // LOAD UNITS
    // =========================================================
    private void loadUnits() {
        categoryUnits.clear();
        if (category != null) {
            categoryUnits.addAll(
                    unitRepository.getUnits(
                            category
                    )
            );
        }
        renderCategory();
        pendingSelectedUnit =
                findInitialUnit();
        unitAdapter.setSelectedUnit(
                pendingSelectedUnit
        );
        unitAdapter.submitList(
                categoryUnits
        );
        btnConfirm.setEnabled(
                pendingSelectedUnit != null
        );
        renderListState(
                categoryUnits
        );
    }
    private void renderCategory() {
        if (category == null) {
            tvCategory.setText(
                    ""
            );
            return;
        }
        tvCategory.setText(
                category.getDisplayName()
        );
    }
    private UnitDefinition findInitialUnit() {
        if (category == null
                || categoryUnits.isEmpty()) {
            return null;
        }
        UnitDefinition currentUnit =
                unitRepository.findBySymbol(
                        category,
                        currentUnitSymbol
                );
        if (currentUnit != null) {
            return currentUnit;
        }
        /*
         * Fallback an toàn:
         *
         * nếu symbol cũ không còn tồn tại,
         * chọn unit đầu tiên của category.
         */
        return categoryUnits.get(
                0
        );
    }
    // =========================================================
    // SEARCH
    // =========================================================
    private void setupSearch() {
        etSearch.addTextChangedListener(
                new TextWatcher() {
                    @Override
                    public void beforeTextChanged(
                            CharSequence text,
                            int start,
                            int count,
                            int after
                    ) {
                        // Không cần xử lý.
                    }
                    @Override
                    public void onTextChanged(
                            CharSequence text,
                            int start,
                            int before,
                            int count
                    ) {
                        filterUnits(
                                text
                        );
                    }
                    @Override
                    public void afterTextChanged(
                            Editable editable
                    ) {
                        // Không cần xử lý.
                    }
                }
        );
    }
    private void filterUnits(
            CharSequence query
    ) {
        String normalizedQuery =
                query == null
                        ? ""
                        : query
                        .toString()
                        .trim()
                        .toLowerCase(
                                Locale.US
                        );
        List<UnitDefinition> filteredUnits =
                new ArrayList<>();
        if (normalizedQuery.isEmpty()) {
            filteredUnits.addAll(
                    categoryUnits
            );
        } else {
            for (UnitDefinition unit
                    : categoryUnits) {
                String name =
                        unit
                                .getName()
                                .toLowerCase(
                                        Locale.US
                                );
                String symbol =
                        unit
                                .getSymbol()
                                .toLowerCase(
                                        Locale.US
                                );
                if (name.contains(
                        normalizedQuery
                ) || symbol.contains(
                        normalizedQuery
                )) {
                    filteredUnits.add(
                            unit
                    );
                }
            }
        }
        unitAdapter.submitList(
                filteredUnits
        );
        renderListState(
                filteredUnits
        );
    }
    // =========================================================
    // ACTIONS
    // =========================================================
    private void setupActions(
            @NonNull View rootView
    ) {
        rootView
                .findViewById(
                        R.id.btnCloseUnitPicker
                )
                .setOnClickListener(view -> {
                    dismiss();
                });
        btnConfirm.setOnClickListener(view -> {
            confirmSelection();
        });
    }
    private void confirmSelection() {
        if (category == null
                || pendingSelectedUnit == null) {
            return;
        }
        Bundle result =
                new Bundle();
        result.putString(
                RESULT_TARGET,
                target
        );
        result.putString(
                RESULT_CATEGORY,
                category.name()
        );
        result.putString(
                RESULT_UNIT_SYMBOL,
                pendingSelectedUnit.getSymbol()
        );
        getParentFragmentManager()
                .setFragmentResult(
                        RESULT_REQUEST_KEY,
                        result
                );
        dismiss();
    }
    // =========================================================
    // VIEW STATE
    // =========================================================
    private void renderListState(
            List<UnitDefinition> visibleUnits
    ) {
        boolean isEmpty =
                visibleUnits == null
                        || visibleUnits.isEmpty();
        recyclerUnits.setVisibility(
                isEmpty
                        ? View.GONE
                        : View.VISIBLE
        );
        tvEmptyUnits.setVisibility(
                isEmpty
                        ? View.VISIBLE
                        : View.GONE
        );
    }
}
