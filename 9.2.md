Bước 9.2 — UnitCategory và dữ liệu offline trong UnitRepository

Spec đã chốt 11 nhóm đơn vị, conversion theo hướng input → base unit → target unit, riêng Temperature xử lý bằng công thức riêng. Tuy nhiên spec không quy định chính xác toàn bộ danh sách unit, nên bước này dùng một bộ dữ liệu MVP ổn định, phổ biến và hoàn toàn offline.  

Ở bước này sửa đúng 2 file:

UnitCategory.java
UnitRepository.java

Chưa sửa:

UnitConverterEngine.java
layout_converter.xml
MainActivity.java

/**
 * Các nhóm đơn vị được hỗ trợ trong Unit Converter.

public enum UnitCategory {
    ANGLE("Angle"),
    AREA("Area"),
    DATA("Data"),

    FORCE("Force"),
    LENGTH("Length"),
    MASS("Mass"),
    SPEED("Speed"),
    TEMPERATURE("Temperature"),
    TIME("Time"),
    VOLUME("Volume");
    private final String displayName;
    UnitCategory(
            String displayName
    ) {
        this.displayName =
                displayName;
    }
    public String getDisplayName() {
        return displayName;
    }
    @Override
    public String toString() {
        return displayName;
    }
}

/**
 * Kho dữ liệu đơn vị hoạt động hoàn toàn offline.
 *
 * Mỗi category có một base unit:
 *
 * Angle      -> Radian
 * Area       -> Square meter
 * Data       -> Byte
 * Energy     -> Joule
 * Force      -> Newton
 * Length     -> Meter
 * Mass       -> Kilogram
 * Speed      -> Meter per second

 * Time       -> Second
 * Volume     -> Liter

 * UnitRepository chỉ lưu dữ liệu.
 * Không thực hiện công thức conversion.

public final class UnitRepository {
    // =========================================================
    // DATA
    // =========================================================
    private final Map<UnitCategory, List<UnitDefinition>>
            unitsByCategory =
            new EnumMap<>(
                    UnitCategory.class
            );
    // =========================================================
    // CONSTRUCTOR
    // =========================================================
    public UnitRepository() {
        initializeUnits();
    }
    // =========================================================
    // PUBLIC API
    // =========================================================
    /**
     * Trả về toàn bộ category theo thứ tự enum.
     */
    public List<UnitCategory> getCategories() {
        List<UnitCategory> categories =
                new ArrayList<>();
        Collections.addAll(
                categories,
                UnitCategory.values()
        );
        return Collections.unmodifiableList(
                categories
        );
    }
    /**
     * Trả về danh sách unit của một category.
     */
    public List<UnitDefinition> getUnits(
            UnitCategory category
    ) {
        if (category == null) {
            return Collections.emptyList();
        }
        List<UnitDefinition> units =
                unitsByCategory.get(
                        category
                );
        if (units == null) {
            return Collections.emptyList();
        }
        return units;
    }
    /**
     * Trả về toàn bộ unit thuộc mọi category.
     */
    public List<UnitDefinition> getAllUnits() {
        List<UnitDefinition> allUnits =
                new ArrayList<>();
        for (UnitCategory category
                : UnitCategory.values()) {
            allUnits.addAll(
                    getUnits(category)
            );
        }
        return Collections.unmodifiableList(
                allUnits
        );
    }
    /**
     * Tìm unit theo:
     *
     * - name
     * - symbol
     * - category
     *
     * Không phân biệt chữ hoa/chữ thường.
     *
     * Query rỗng trả về toàn bộ unit.
     */
    public List<UnitDefinition> searchUnits(
            String query
    ) {
        String normalizedQuery =
                query == null
                        ? ""
                        : query
                        .trim()
                        .toLowerCase(
                                Locale.US
                        );
        if (normalizedQuery.isEmpty()) {
            return getAllUnits();
        }
        List<UnitDefinition> results =
                new ArrayList<>();
        for (UnitDefinition unit
                : getAllUnits()) {
            String name =
                    unit
                            .getName()
                            .toLowerCase(
                                    Locale.US
                            );
            String symbol =
                    unit
                            .getSymbol()
                            .toLowerCase(
                                    Locale.US
                            );
            String category =
                    unit
                            .getCategory()
                            .getDisplayName()
                            .toLowerCase(
                                    Locale.US
                            );
            if (name.contains(normalizedQuery)
                    || symbol.contains(normalizedQuery)
                    || category.contains(normalizedQuery)) {
                results.add(
                        unit
                );
            }
        }
        return Collections.unmodifiableList(
                results
        );
    }
    /**
     * Tìm một unit theo category và symbol.
     *
     * Method này sẽ hữu ích khi Converter cần khôi phục
     * lựa chọn mặc định hoặc state đã lưu.
     */
    public UnitDefinition findBySymbol(
            UnitCategory category,
            String symbol
    ) {
        if (category == null
                || symbol == null
                || symbol.trim().isEmpty()) {
            return null;
        }
        for (UnitDefinition unit
                : getUnits(category)) {
            if (unit
                    .getSymbol()
                    .equalsIgnoreCase(
                            symbol.trim()
                    )) {
                return unit;
            }
        }
        return null;
    }

4. Lưu ý về Temperature

Ba unit:

Celsius
Fahrenheit
Kelvin

đều đang có:

factorToBase = 1.0;

Đây không phải công thức chuyển đổi.

Temperature bắt buộc xử lý riêng trong UnitConverterEngine:

°C ↔ °F ↔ K

vì có cả scale và offset.
