Bước 9.6 — Tạo ConverterController

Theo kiến trúc đã chốt, UI chỉ nhận input và hiển thị output; việc chuyển đổi tiếp tục nằm trong UnitConverterEngine, còn danh sách đơn vị nằm trong UnitRepository.   Bộ dữ liệu hiện đã có đủ 11 category và API như getUnits() तथा findBySymbol().  

Bước này tạo controller có sẵn:

Realtime conversion
Swap From ↔ To
Đổi category
Đổi From unit
Đổi To unit
Điểm nối cho Unit Picker

Controller chưa được nối vào MainActivity ở lượt này.

package com.example.calculator.ui.controller;
import android.text.Editable;
import android.text.TextWatcher;
import android.view.View;
import android.widget.EditText;
import android.widget.TextView;
import com.example.calculator.R;
import com.example.calculator.converter.UnitConverterEngine;
import com.example.calculator.converter.UnitRepository;
import com.example.calculator.model.UnitCategory;
import com.example.calculator.model.UnitDefinition;
import java.util.List;
/**
 * Điều phối toàn bộ giao diện Unit Converter.
 *
 * Trách nhiệm:
 *
 * - Bind Converter Views
 * - Quản lý category hiện tại
 * - Quản lý From/To unit
 * - Theo dõi input
 * - Gọi UnitConverterEngine
 * - Render kết quả
 * - Swap hai unit
 * - Gửi yêu cầu mở picker qua callback
 *
 * Không chứa conversion formula.
 */
public final class ConverterController {
    // =========================================================
    // PICKER CALLBACK
    // =========================================================
    /**
     * Callback để MainActivity hoặc UnitPicker coordinator
     * mở category/unit picker.
     *
     * Unit Picker thật sẽ được triển khai ở bước sau.
     */
    public interface PickerRequestListener {
        void onCategoryPickerRequested(
                UnitCategory currentCategory
        );
        void onFromUnitPickerRequested(
                UnitCategory category,
                UnitDefinition currentUnit
        );
        void onToUnitPickerRequested(
                UnitCategory category,
                UnitDefinition currentUnit
        );
    }
    // =========================================================
    // DEPENDENCIES
    // =========================================================
    private final View rootView;
    private final UnitRepository unitRepository;
    private final UnitConverterEngine converterEngine;
    // =========================================================
    // CALLBACK
    // =========================================================
    private PickerRequestListener pickerRequestListener;
    // =========================================================
    // VIEWS
    // =========================================================
    private View categorySelector;
    private View fromUnitSelector;
    private View toUnitSelector;
    private View swapButton;
    private EditText etInput;
    private TextView tvCategory;
    private TextView tvFromUnitName;
    private TextView tvFromUnitSymbol;
    private TextView tvToUnitName;
    private TextView tvToUnitSymbol;
    private TextView tvResult;
    // =========================================================
    // SELECTION STATE
    // =========================================================
    private UnitCategory selectedCategory;
    private UnitDefinition fromUnit;
    private UnitDefinition toUnit;
    // =========================================================
    // CONSTRUCTOR
    // =========================================================
    public ConverterController(
            View rootView,
            UnitRepository unitRepository,
            UnitConverterEngine converterEngine
    ) {
        if (rootView == null) {
            throw new IllegalArgumentException(
                    "Converter root view cannot be null."
            );
        }
        if (unitRepository == null) {
            throw new IllegalArgumentException(
                    "UnitRepository cannot be null."
            );
        }
        if (converterEngine == null) {
            throw new IllegalArgumentException(
                    "UnitConverterEngine cannot be null."
            );
        }
        this.rootView =
                rootView;
        this.unitRepository =
                unitRepository;
        this.converterEngine =
                converterEngine;
    }
    // =========================================================
    // PUBLIC SETUP
    // =========================================================
    /**
     * Bind Views, setup listeners và render mặc định.
     *
     * Chỉ gọi một lần sau khi layout_converter.xml
     * được inflate.
     */
    public void setup() {
        bindViews();
        setupDefaultSelection();
        setupInputListener();
        setupActions();
        renderSelection();
        convertCurrentInput();
    }
    /**
     * Gắn callback mở picker.
     *
     * Có thể gọi trước hoặc sau setup().
     */
    public void setPickerRequestListener(
            PickerRequestListener listener
    ) {
        pickerRequestListener =
                listener;
    }
    // =========================================================
    // PUBLIC STATE
    // =========================================================
    public UnitCategory getSelectedCategory() {
        return selectedCategory;
    }
    public UnitDefinition getFromUnit() {
        return fromUnit;
    }
    public UnitDefinition getToUnit() {
        return toUnit;
    }
    // =========================================================
    // PUBLIC SELECTION API
    // =========================================================
    /**
     * Được gọi sau khi user chọn category.
     *
     * Khi đổi category:
     *
     * - From/To cũ bị thay thế;
     * - chọn cặp unit mặc định của category mới;
     * - conversion được tính lại.
     */
    public void selectCategory(
            UnitCategory category
    ) {
        if (category == null) {
            return;
        }
        List<UnitDefinition> units =
                unitRepository.getUnits(
                        category
                );
        if (units.isEmpty()) {
            return;
        }
        selectedCategory =
                category;
        selectDefaultUnitsForCategory(
                category
        );
        renderSelection();
        convertCurrentInput();
    }
    /**
     * Được gọi sau khi user chọn From unit.
     */
    public void selectFromUnit(
            UnitDefinition unit
    ) {
        if (!isValidUnitForCurrentCategory(
                unit
        )) {
            return;
        }
        fromUnit =
                unit;
        renderSelection();
        convertCurrentInput();
    }
    /**
     * Được gọi sau khi user chọn To unit.
     */
    public void selectToUnit(
            UnitDefinition unit
    ) {
        if (!isValidUnitForCurrentCategory(
                unit
        )) {
            return;
        }
        toUnit =
                unit;
        renderSelection();
        convertCurrentInput();
    }
    /**
     * Yêu cầu controller tính lại từ input hiện tại.
     */
    public void refreshConversion() {
        convertCurrentInput();
    }
    // =========================================================
    // BIND VIEWS
    // =========================================================
    private void bindViews() {
        categorySelector =
                requireView(
                        R.id.converterCategorySelector
                );
        fromUnitSelector =
                requireView(
                        R.id.converterFromUnitSelector
                );
        toUnitSelector =
                requireView(
                        R.id.converterToUnitSelector
                );
        swapButton =
                requireView(
                        R.id.btnSwapConverterUnits
                );
        etInput =
                requireView(
                        R.id.etConverterInput
                );
        tvCategory =
                requireView(
                        R.id.tvConverterCategory
                );
        tvFromUnitName =
                requireView(
                        R.id.tvConverterFromUnitName
                );
        tvFromUnitSymbol =
                requireView(
                        R.id.tvConverterFromUnitSymbol
                );
        tvToUnitName =
                requireView(
                        R.id.tvConverterToUnitName
                );
        tvToUnitSymbol =
                requireView(
                        R.id.tvConverterToUnitSymbol
                );
        tvResult =
                requireView(
                        R.id.tvConverterResult
                );
    }
    private <T extends View> T requireView(
            int viewId
    ) {
        T view =
                rootView.findViewById(
                        viewId
                );
        if (view == null) {
            throw new IllegalStateException(
                    "Missing Converter View ID: "
                            + viewId
            );
        }
        return view;
    }
    // =========================================================
    // DEFAULT STATE
    // =========================================================
    private void setupDefaultSelection() {
        selectedCategory =
                UnitCategory.LENGTH;
        selectDefaultUnitsForCategory(
                selectedCategory
        );
    }
    /**
     * Chọn cặp mặc định hợp lý cho từng category.
     */
    private void selectDefaultUnitsForCategory(
            UnitCategory category
    ) {
        if (category == UnitCategory.LENGTH) {
            fromUnit =
                    unitRepository.findBySymbol(
                            UnitCategory.LENGTH,
                            "km"
                    );
            toUnit =
                    unitRepository.findBySymbol(
                            UnitCategory.LENGTH,
                            "m"
                    );
            if (fromUnit != null
                    && toUnit != null) {
                return;
            }
        }
        if (category == UnitCategory.TEMPERATURE) {
            fromUnit =
                    unitRepository.findBySymbol(
                            UnitCategory.TEMPERATURE,
                            "°C"
                    );
            toUnit =
                    unitRepository.findBySymbol(
                            UnitCategory.TEMPERATURE,
                            "°F"
                    );
            if (fromUnit != null
                    && toUnit != null) {
                return;
            }
        }
        /*
         * Các category khác:
         *
         * unit đầu tiên → unit thứ hai.
         */
        List<UnitDefinition> units =
                unitRepository.getUnits(
                        category
                );
        if (units.isEmpty()) {
            fromUnit =
                    null;
            toUnit =
                    null;
            return;
        }
        fromUnit =
                units.get(0);
        if (units.size() >= 2) {
            toUnit =
                    units.get(1);
        } else {
            toUnit =
                    units.get(0);
        }
    }
    // =========================================================
    // INPUT LISTENER
    // =========================================================
    private void setupInputListener() {
        etInput.addTextChangedListener(
                new TextWatcher() {
                    @Override
                    public void beforeTextChanged(
                            CharSequence text,
                            int start,
                            int count,
                            int after
                    ) {
                        // Không cần xử lý.
                    }
                    @Override
                    public void onTextChanged(
                            CharSequence text,
                            int start,
                            int before,
                            int count
                    ) {
                        convertCurrentInput();
                    }
                    @Override
                    public void afterTextChanged(
                            Editable editable
                    ) {
                        // Không cần xử lý.
                    }
                }
        );
    }
    // =========================================================
    // ACTIONS
    // =========================================================
    private void setupActions() {
        categorySelector.setOnClickListener(view -> {
            if (pickerRequestListener == null) {
                return;
            }
            pickerRequestListener
                    .onCategoryPickerRequested(
                            selectedCategory
                    );
        });
        fromUnitSelector.setOnClickListener(view -> {
            if (pickerRequestListener == null) {
                return;
            }
            pickerRequestListener
                    .onFromUnitPickerRequested(
                            selectedCategory,
                            fromUnit
                    );
        });
        toUnitSelector.setOnClickListener(view -> {
            if (pickerRequestListener == null) {
                return;
            }
            pickerRequestListener
                    .onToUnitPickerRequested(
                            selectedCategory,
                            toUnit
                    );
        });
        swapButton.setOnClickListener(view -> {
            swapUnits();
        });
    }
    // =========================================================
    // SWAP
    // =========================================================
    private void swapUnits() {
        if (fromUnit == null
                || toUnit == null) {
            return;
        }
        UnitDefinition previousFrom =
                fromUnit;
        fromUnit =
                toUnit;
        toUnit =
                previousFrom;
        renderSelection();
        convertCurrentInput();
    }
    // =========================================================
    // CONVERSION
    // =========================================================
    private void convertCurrentInput() {
        if (etInput == null
                || tvResult == null) {
            return;
        }
        String inputText =
                etInput
                        .getText()
                        .toString()
                        .trim();
        /*
         * Partial input hợp lệ trong lúc user đang gõ.
         *
         * Không hiển thị Error cho:
         *
         * ""
         * "-"
         * "."
         * "-."
         */
        if (inputText.isEmpty()
                || "-".equals(inputText)
                || ".".equals(inputText)
                || "-.".equals(inputText)) {
            tvResult.setText(
                    ""
            );
            return;
        }
        /*
         * Hỗ trợ cả dấu chấm và dấu phẩy thập phân.
         */
        String normalizedInput =
                inputText.replace(
                        ',',
                        '.'
                );
        double inputValue;
        try {
            inputValue =
                    Double.parseDouble(
                            normalizedInput
                    );
        } catch (NumberFormatException exception) {
            tvResult.setText(
                    "Error"
            );
            return;
        }
        UnitConverterEngine.ConversionResult result =
                converterEngine.convert(
                        inputValue,
                        fromUnit,
                        toUnit
                );
        if (result.isSuccess()) {
            tvResult.setText(
                    result.getFormattedValue()
            );
        } else {
            tvResult.setText(
                    "Error"
            );
        }
    }
    // =========================================================
    // RENDER
    // =========================================================
    private void renderSelection() {
        renderCategory();
        renderFromUnit();
        renderToUnit();
    }
    private void renderCategory() {
        if (selectedCategory == null) {
            tvCategory.setText(
                    ""
            );
            return;
        }
        tvCategory.setText(
                selectedCategory.getDisplayName()
        );
    }
    private void renderFromUnit() {
        if (fromUnit == null) {
            tvFromUnitName.setText(
                    ""
            );
            tvFromUnitSymbol.setText(
                    ""
            );
            return;
        }
        tvFromUnitName.setText(
                fromUnit.getName()
        );
        tvFromUnitSymbol.setText(
                fromUnit.getSymbol()
        );
    }
    private void renderToUnit() {
        if (toUnit == null) {
            tvToUnitName.setText(
                    ""
            );
            tvToUnitSymbol.setText(
                    ""
            );
            return;
        }
        tvToUnitName.setText(
                toUnit.getName()
        );
        tvToUnitSymbol.setText(
                toUnit.getSymbol()
        );
    }
    // =========================================================
    // VALIDATION
    // =========================================================
    private boolean isValidUnitForCurrentCategory(
            UnitDefinition unit
    ) {
        return unit != null
                && selectedCategory != null
                && unit.getCategory()
                == selectedCategory;
    }
}
