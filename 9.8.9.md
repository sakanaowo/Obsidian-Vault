3. Thêm showCategoryPicker()

Đặt method này ngay sau setupConverterController():

// =========================================================
// CONVERTER CATEGORY PICKER
// =========================================================
/**
 * Hiển thị danh sách 11 conversion category.
 *
 * Khi user chọn một category:
 *
 * - cập nhật category;
 * - chọn lại From/To unit mặc định;
 * - tính lại kết quả.
 */
private void showCategoryPicker(
        UnitCategory currentCategory
) {
    UnitCategory[] categories =
            UnitCategory.values();
    String[] categoryNames =
            new String[
                    categories.length
            ];
    for (int index = 0;
         index < categories.length;
         index++) {
        categoryNames[index] =
                categories[index]
                        .getDisplayName();
    }
    int checkedIndex =
            findCategoryIndex(
                    categories,
                    currentCategory
            );
    AlertDialog dialog =
            new AlertDialog.Builder(
                    this
            )
                    .setTitle(
                            R.string.converter_open_category
                    )
                    .setSingleChoiceItems(
                            categoryNames,
                            checkedIndex,
                            null
                    )
                    .setNegativeButton(
                            R.string.action_cancel,
                            null
                    )
                    .create();
    dialog.setOnShowListener(
            ignored -> {
                dialog
                        .getListView()
                        .setOnItemClickListener(
                                (parent,
                                 view,
                                 position,
                                 id) -> {
                                    if (position < 0
                                            || position
                                            >= categories.length) {
                                        return;
                                    }
                                    converterController
                                            .selectCategory(
                                                    categories[position]
                                            );
                                    dialog.dismiss();
                                }
                        );
            }
    );
    dialog.show();
}
/**
 * Tìm vị trí checkmark của category hiện tại.
 */
private int findCategoryIndex(
        UnitCategory[] categories,
        UnitCategory currentCategory
) {
    if (categories == null
            || categories.length == 0
            || currentCategory == null) {
        return -1;
    }
    for (int index = 0;
         index < categories.length;
         index++) {
        if (categories[index]
                == currentCategory) {
            return index;
        }
    }
    return -1;
}

Bước 9.9 — Nối Unit Picker cho From và To

Bước 9.8 đã cho phép đổi một trong 11 category. Bây giờ hai selector From unit và To unit sẽ mở danh sách đơn vị thuộc category hiện tại, sau đó gọi:

converterController.selectFromUnit(selectedUnit);

hoặc:

converterController.selectToUnit(selectedUnit);

Unit Picker đầy đủ theo thiết kế còn cần tìm kiếm và giao diện Bottom Sheet; bước này triển khai phiên bản danh sách đơn giản trước để hoàn thiện flow chuyển đổi.

4. Thêm showUnitPicker()

Đặt method này ngay dưới nhóm Category Picker:

// =========================================================
// CONVERTER UNIT PICKER
// =========================================================
/**
 * Hiển thị các unit thuộc category hiện tại.
 *
 * Ví dụ:
 *
 * LENGTH
 *   ↓
 * Meter
 * Kilometer
 * Centimeter
 * ...
 *
 * @param category     category hiện tại
 * @param currentUnit  unit đang được chọn
 * @param selectingFrom true nếu chọn From,
 *                      false nếu chọn To
 */
private void showUnitPicker(
        UnitCategory category,
        UnitDefinition currentUnit,
        boolean selectingFrom
) {
    if (category == null
            || unitRepository == null) {
        return;
    }
    List<UnitDefinition> units =
            unitRepository.getUnits(
                    category
            );
    if (units == null
            || units.isEmpty()) {
        return;
    }
    String[] unitLabels =
            new String[
                    units.size()
            ];
    for (int index = 0;
         index < units.size();
         index++) {
        UnitDefinition unit =
                units.get(index);
        unitLabels[index] =
                unit.getName()
                        + "  ("
                        + unit.getSymbol()
                        + ")";
    }
    int checkedIndex =
            findUnitIndex(
                    units,
                    currentUnit
            );
    int titleResource =
            selectingFrom
                    ? R.string.converter_open_from_unit
                    : R.string.converter_open_to_unit;
    AlertDialog dialog =
            new AlertDialog.Builder(
                    this
            )
                    .setTitle(
                            titleResource
                    )
                    .setSingleChoiceItems(
                            unitLabels,
                            checkedIndex,
                            null
                    )
                    .setNegativeButton(
                            R.string.action_cancel,
                            null
                    )
                    .create();
    dialog.setOnShowListener(
            ignored -> {
                dialog
                        .getListView()
                        .setOnItemClickListener(
                                (parent,
                                 view,
                                 position,
                                 id) -> {
                                    if (position < 0
                                            || position
                                            >= units.size()) {
                                        return;
                                    }
                                    UnitDefinition selectedUnit =
                                            units.get(
                                                    position
                                            );
                                    if (selectingFrom) {
                                        converterController
                                                .selectFromUnit(
                                                        selectedUnit
                                                );
                                    } else {
                                        converterController
                                                .selectToUnit(
                                                        selectedUnit
                                                );
                                    }
                                    dialog.dismiss();
                                }
                        );
            }
    );
    dialog.show();
}

5. Thêm findUnitIndex()

Đặt ngay dưới showUnitPicker():

/**
 * Tìm vị trí checkmark của unit hiện tại.
 */
private int findUnitIndex(
        List<UnitDefinition> units,
        UnitDefinition currentUnit
) {
    if (units == null
            || units.isEmpty()
            || currentUnit == null) {
        return -1;
    }
    for (int index = 0;
         index < units.size();
         index++) {
        if (currentUnit.equals(
                units.get(index)
        )) {
            return index;
        }
    }
    return -1;
}
