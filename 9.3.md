Bước 9.3 — Implement UnitConverterEngine cho các nhóm dùng hệ số

Bước 9.2 đã tạo dữ liệu offline và quy định base unit cho từng category. Luồng chuyển đổi đã chốt là:

input
→ nhân factorToBase của From unit
→ giá trị tại base unit
→ chia factorToBase của To unit
→ result

Riêng TEMPERATURE chưa xử lý trong lượt này vì cần cả hệ số và offset; phần đó thuộc Bước 9.4.

package com.example.calculator.converter;
import com.example.calculator.model.UnitCategory;
import com.example.calculator.model.UnitDefinition;
import com.example.calculator.utils.NumberFormatter;
/**
 * Xử lý business logic của Unit Converter.
 *
 * Engine không phụ thuộc Android UI.
 *
 * Với category thông thường:
 *
 * input
 *      ↓
 * input × from.factorToBase
 *      ↓
 * base value
 *      ↓
 * base value ÷ to.factorToBase
 *      ↓
 * result
 *
 * Temperature sẽ được bổ sung riêng ở Bước 9.4
 * vì cần cả scale và offset.
 */
public final class UnitConverterEngine {
    // =========================================================
    // PUBLIC API
    // =========================================================
    /**
     * Chuyển đổi một giá trị giữa hai unit cùng category.
     *
     * Không throw exception ra UI.
     * Mọi lỗi được trả về bằng ConversionResult.
     */
    public ConversionResult convert(
            double inputValue,
            UnitDefinition fromUnit,
            UnitDefinition toUnit
    ) {
        /*
         * Input phải là số hữu hạn.
         */
        if (!isFinite(inputValue)) {
            return ConversionResult.error(
                    "Input value must be a finite number."
            );
        }
        /*
         * Hai unit bắt buộc phải tồn tại.
         */
        if (fromUnit == null) {
            return ConversionResult.error(
                    "From unit cannot be null."
            );
        }
        if (toUnit == null) {
            return ConversionResult.error(
                    "To unit cannot be null."
            );
        }
        /*
         * Không cho convert giữa hai category khác nhau.
         *
         * Ví dụ không hợp lệ:
         *
         * Kilometer → Kilogram
         */
        if (fromUnit.getCategory()
                != toUnit.getCategory()) {
            return ConversionResult.error(
                    "Units must belong to the same category."
            );
        }
        /*
         * Temperature chưa dùng factorToBase.
         *
         * Công thức °C / °F / K sẽ được thêm
         * trong Bước 9.4.
         */
        if (fromUnit.getCategory()
                == UnitCategory.TEMPERATURE) {
            return ConversionResult.error(
                    "Temperature conversion is not implemented yet."
            );
        }
        /*
         * Cùng một unit:
         *
         * 25 m → m
         *
         * giữ nguyên 25.
         */
        if (fromUnit.equals(toUnit)) {
            return ConversionResult.success(
                    inputValue
            );
        }
        /*
         * Bước 1:
         *
         * From unit → base unit
         *
         * Ví dụ:
         *
         * 1 km × 1000
         * = 1000 m
         */
        double baseValue =
                inputValue
                        * fromUnit.getFactorToBase();
        if (!isFinite(baseValue)) {
            return ConversionResult.error(
                    "Conversion result is outside the supported range."
            );
        }
        /*
         * Bước 2:
         *
         * Base unit → To unit
         *
         * Ví dụ:
         *
         * 1000 m ÷ 1
         * = 1000 m
         */
        double convertedValue =
                baseValue
                        / toUnit.getFactorToBase();
        if (!isFinite(convertedValue)) {
            return ConversionResult.error(
                    "Conversion result is outside the supported range."
            );
        }
        return ConversionResult.success(
                convertedValue
        );
    }
    // =========================================================
    // VALIDATION
    // =========================================================
    private boolean isFinite(
            double value
    ) {
        return !Double.isNaN(value)
                && !Double.isInfinite(value);
    }
    // =========================================================
    // RESULT MODEL
    // =========================================================
    /**
     * Kết quả trả về cho UI Controller.
     *
     * UI không cần catch exception và không tự format số.
     */
    public static final class ConversionResult {
        private final boolean success;
        private final double value;
        private final String formattedValue;
        private final String errorMessage;
        private ConversionResult(
                boolean success,
                double value,
                String formattedValue,
                String errorMessage
        ) {
            this.success =
                    success;
            this.value =
                    value;
            this.formattedValue =
                    formattedValue;
            this.errorMessage =
                    errorMessage;
        }
        private static ConversionResult success(
                double value
        ) {
            return new ConversionResult(
                    true,
                    value,
                    NumberFormatter.format(
                            value
                    ),
                    null
            );
        }
        private static ConversionResult error(
                String errorMessage
        ) {
            return new ConversionResult(
                    false,
                    Double.NaN,
                    "Error",
                    errorMessage
            );
        }
        public boolean isSuccess() {
            return success;
        }
        /**
         * Chỉ đọc khi isSuccess() == true.
         */
        public double getValue() {
            return value;
        }
        /**
         * Chuỗi đã được NumberFormatter xử lý.
         *
         * Ví dụ:
         *
         * 1000.0 → "1000"
         * 0.30000000000000004 → "0.3"
         */
        public String getFormattedValue() {
            return formattedValue;
        }
        /**
         * null khi conversion thành công.
         */
        public String getErrorMessage() {
            return errorMessage;
        }
    }
}

Ví dụ hoạt động

Length

UnitRepository repository =
        new UnitRepository();
UnitConverterEngine engine =
        new UnitConverterEngine();
UnitDefinition kilometer =
        repository.findBySymbol(
                UnitCategory.LENGTH,
                "km"
        );
UnitDefinition meter =
        repository.findBySymbol(
                UnitCategory.LENGTH,
                "m"
        );
UnitConverterEngine.ConversionResult result =
        engine.convert(
                1.0,
                kilometer,
                meter
        );

result.isSuccess()       = true
result.getValue()        = 1000.0
result.getFormattedValue() = "1000"
