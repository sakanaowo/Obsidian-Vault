Bước 9.10D — Thêm CategoryPickerBottomSheet

Đúng theo spec, phần chọn đơn vị cần có category selector, danh sách, chọn và xác nhận; Category không nên tiếp tục dùng AlertDialog.   Kiến trúc hiện tại cũng đã dùng Bottom Sheet cho Unit Picker, nên Category Picker sẽ dùng cùng pattern để UI đồng nhất.  

Sau bước này:

Category
    ↓
CategoryPickerBottomSheet
    ↓
Chọn category tạm thời
    ↓
Confirm
    ↓
ConverterController.selectCategory(...)

Nhấn Close hoặc vuốt sheet xuống sẽ không thay đổi category.

package com.example.calculator.ui.adapter;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.ImageView;
import android.widget.TextView;
import androidx.annotation.NonNull;
import androidx.recyclerview.widget.RecyclerView;
import com.example.calculator.R;
import com.example.calculator.model.UnitCategory;
import java.util.ArrayList;
import java.util.List;
/**
 * Adapter hiển thị danh sách UnitCategory.
 *
 * Adapter chỉ quản lý UI selection.
 * Không trực tiếp thay đổi ConverterController.
 */
public final class CategoryAdapter
        extends RecyclerView.Adapter<
        CategoryAdapter.CategoryViewHolder> {
    // =========================================================
    // CALLBACK
    // =========================================================
    public interface OnCategorySelectedListener {
        void onCategorySelected(
                UnitCategory category
        );
    }
    // =========================================================
    // DATA
    // =========================================================
    private final List<UnitCategory> categories =
            new ArrayList<>();
    private final OnCategorySelectedListener
            categorySelectedListener;
    private UnitCategory selectedCategory;
    // =========================================================
    // CONSTRUCTOR
    // =========================================================
    public CategoryAdapter(
            OnCategorySelectedListener listener
    ) {
        categorySelectedListener =
                listener;
    }
    // =========================================================
    // CREATE VIEW HOLDER
    // =========================================================
    @NonNull
    @Override
    public CategoryViewHolder onCreateViewHolder(
            @NonNull ViewGroup parent,
            int viewType
    ) {
        View itemView =
                LayoutInflater
                        .from(parent.getContext())
                        .inflate(
                                R.layout.item_category,
                                parent,
                                false
                        );
        return new CategoryViewHolder(
                itemView
        );
    }
    // =========================================================
    // BIND
    // =========================================================
    @Override
    public void onBindViewHolder(
            @NonNull CategoryViewHolder holder,
            int position
    ) {
        UnitCategory category =
                categories.get(position);
        boolean selected =
                category == selectedCategory;
        holder.bind(
                category,
                selected,
                this::selectCategory
        );
    }
    @Override
    public int getItemCount() {
        return categories.size();
    }
    // =========================================================
    // PUBLIC API
    // =========================================================
    public void submitList(
            List<UnitCategory> newCategories
    ) {
        categories.clear();
        if (newCategories != null) {
            categories.addAll(
                    newCategories
            );
        }
        notifyDataSetChanged();
    }
    public void setSelectedCategory(
            UnitCategory category
    ) {
        selectedCategory =
                category;
        notifyDataSetChanged();
    }
    public UnitCategory getSelectedCategory() {
        return selectedCategory;
    }
    // =========================================================
    // SELECTION
    // =========================================================
    private void selectCategory(
            UnitCategory category
    ) {
        if (category == null
                || category == selectedCategory) {
            return;
        }
        selectedCategory =
                category;
        notifyDataSetChanged();
        if (categorySelectedListener != null) {
            categorySelectedListener
                    .onCategorySelected(
                            category
                    );
        }
    }
    // =========================================================
    // VIEW HOLDER
    // =========================================================
    static final class CategoryViewHolder
            extends RecyclerView.ViewHolder {
        private final TextView tvCategoryName;
        private final ImageView ivCategorySelected;
        CategoryViewHolder(
                @NonNull View itemView
        ) {
            super(itemView);
            tvCategoryName =
                    itemView.findViewById(
                            R.id.tvCategoryName
                    );
            ivCategorySelected =
                    itemView.findViewById(
                            R.id.ivCategorySelected
                    );
        }
        void bind(
                UnitCategory category,
                boolean selected,
                OnCategorySelectedListener listener
        ) {
            tvCategoryName.setText(
                    category.getDisplayName()
            );
            ivCategorySelected.setVisibility(
                    selected
                            ? View.VISIBLE
                            : View.GONE
            );
            itemView.setOnClickListener(view -> {
                if (listener != null) {
                    listener.onCategorySelected(
                            category
                    );
                }
            });
        }
    }
}

package com.example.calculator.ui;
import android.app.Dialog;
import android.os.Bundle;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.Button;
import android.widget.FrameLayout;
import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import androidx.recyclerview.widget.LinearLayoutManager;
import androidx.recyclerview.widget.RecyclerView;
import com.example.calculator.R;
import com.example.calculator.model.UnitCategory;
import com.example.calculator.ui.adapter.CategoryAdapter;
import com.google.android.material.bottomsheet.BottomSheetBehavior;
import com.google.android.material.bottomsheet.BottomSheetDialog;
import com.google.android.material.bottomsheet.BottomSheetDialogFragment;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
/**
 * Bottom Sheet chọn conversion category.
 *
 * Luồng:
 *
 * current category
 *      ↓
 * chọn category tạm thời
 *      ↓
 * Confirm
 *      ↓
 * Fragment Result
 *
 * Close hoặc swipe down không gửi kết quả.
 */
public class CategoryPickerBottomSheet
        extends BottomSheetDialogFragment {
    // =========================================================
    // FRAGMENT RESULT
    // =========================================================
    public static final String RESULT_REQUEST_KEY =
            "category_picker_result";
    public static final String RESULT_CATEGORY =
            "category_picker_result_category";
    // =========================================================
    // ARGUMENT
    // =========================================================
    private static final String ARG_CURRENT_CATEGORY =
            "category_picker_current_category";
    // =========================================================
    // STATE
    // =========================================================
    private UnitCategory currentCategory;
    private UnitCategory pendingSelectedCategory;
    // =========================================================
    // UI
    // =========================================================
    private RecyclerView recyclerCategories;
    private Button btnConfirmCategory;
    private CategoryAdapter categoryAdapter;
    // =========================================================
    // FACTORY
    // =========================================================
    public static CategoryPickerBottomSheet newInstance(
            UnitCategory currentCategory
    ) {
        CategoryPickerBottomSheet bottomSheet =
                new CategoryPickerBottomSheet();
        Bundle arguments =
                new Bundle();
        if (currentCategory != null) {
            arguments.putString(
                    ARG_CURRENT_CATEGORY,
                    currentCategory.name()
            );
        }
        bottomSheet.setArguments(
                arguments
        );
        return bottomSheet;
    }
    // =========================================================
    // LIFECYCLE
    // =========================================================
    @Override
    public void onCreate(
            @Nullable Bundle savedInstanceState
    ) {
        super.onCreate(
                savedInstanceState
        );
        readArguments();
    }
    @Nullable
    @Override
    public View onCreateView(
            @NonNull LayoutInflater inflater,
            @Nullable ViewGroup container,
            @Nullable Bundle savedInstanceState
    ) {
        return inflater.inflate(
                R.layout.bottom_sheet_category_picker,
                container,
                false
        );
    }
    @Override
    public void onViewCreated(
            @NonNull View view,
            @Nullable Bundle savedInstanceState
    ) {
        super.onViewCreated(
                view,
                savedInstanceState
        );
        bindViews(
                view
        );
        setupRecyclerView();
        setupActions(
                view
        );
        loadCategories();
    }
    @Override
    public void onStart() {
        super.onStart();
        Dialog dialog =
                getDialog();
        if (!(dialog instanceof BottomSheetDialog)) {
            return;
        }
        BottomSheetDialog bottomSheetDialog =
                (BottomSheetDialog) dialog;
        FrameLayout sheet =
                bottomSheetDialog.findViewById(
                        com.google.android.material.R.id
                                .design_bottom_sheet
                );
        if (sheet == null) {
            return;
        }
        ViewGroup.LayoutParams layoutParams =
                sheet.getLayoutParams();
        layoutParams.height =
                ViewGroup.LayoutParams.MATCH_PARENT;
        sheet.setLayoutParams(
                layoutParams
        );
        BottomSheetBehavior<FrameLayout> behavior =
                BottomSheetBehavior.from(
                        sheet
                );
        behavior.setSkipCollapsed(
                true
        );
        behavior.setState(
                BottomSheetBehavior.STATE_EXPANDED
        );
    }
    @Override
    public void onDestroyView() {
        super.onDestroyView();
        recyclerCategories =
                null;
        btnConfirmCategory =
                null;
        categoryAdapter =
                null;
    }
    // =========================================================
    // ARGUMENTS
    // =========================================================
    private void readArguments() {
        Bundle arguments =
                getArguments();
        if (arguments == null) {
            currentCategory =
                    UnitCategory.LENGTH;
            return;
        }
        String categoryName =
                arguments.getString(
                        ARG_CURRENT_CATEGORY
                );
        currentCategory =
                parseCategory(
                        categoryName
                );
        if (currentCategory == null) {
            currentCategory =
                    UnitCategory.LENGTH;
        }
    }
    private UnitCategory parseCategory(
            String categoryName
    ) {
        if (categoryName == null
                || categoryName.trim().isEmpty()) {
            return null;
        }
        try {
            return UnitCategory.valueOf(
                    categoryName
            );
        } catch (IllegalArgumentException exception) {
            return null;
        }
    }
    // =========================================================
    // BIND
    // =========================================================
    private void bindViews(
            @NonNull View rootView
    ) {
        recyclerCategories =
                requireView(
                        rootView,
                        R.id.recyclerCategories
                );
        btnConfirmCategory =
                requireView(
                        rootView,
                        R.id.btnConfirmCategory
                );
    }
    private <T extends View> T requireView(
            View rootView,
            int viewId
    ) {
        T view =
                rootView.findViewById(
                        viewId
                );
        if (view == null) {
            throw new IllegalStateException(
                    "Missing Category Picker View ID: "
                            + viewId
            );
        }
        return view;
    }
    // =========================================================
    // RECYCLER VIEW
    // =========================================================
    private void setupRecyclerView() {
        recyclerCategories.setLayoutManager(
                new LinearLayoutManager(
                        requireContext()
                )
        );
        categoryAdapter =
                new CategoryAdapter(
                        this::onTemporaryCategorySelected
                );
        recyclerCategories.setAdapter(
                categoryAdapter
        );
    }
    private void onTemporaryCategorySelected(
            UnitCategory category
    ) {
        if (category == null) {
            return;
        }
        pendingSelectedCategory =
                category;
        btnConfirmCategory.setEnabled(
                true
        );
    }
    // =========================================================
    // LOAD
    // =========================================================
    private void loadCategories() {
        List<UnitCategory> categories =
                new ArrayList<>();
        Collections.addAll(
                categories,
                UnitCategory.values()
        );
        pendingSelectedCategory =
                currentCategory;
        categoryAdapter.setSelectedCategory(
                pendingSelectedCategory
        );
        categoryAdapter.submitList(
                categories
        );
        btnConfirmCategory.setEnabled(
                pendingSelectedCategory != null
        );
    }
    // =========================================================
    // ACTIONS
    // =========================================================
    private void setupActions(
            @NonNull View rootView
    ) {
        rootView
                .findViewById(
                        R.id.btnCloseCategoryPicker
                )
                .setOnClickListener(view -> {
                    dismiss();
                });
        btnConfirmCategory.setOnClickListener(view -> {
            confirmSelection();
        });
    }
    private void confirmSelection() {
        if (pendingSelectedCategory == null) {
            return;
        }
        Bundle result =
                new Bundle();
        result.putString(
                RESULT_CATEGORY,
                pendingSelectedCategory.name()
        );
        getParentFragmentManager()
                .setFragmentResult(
                        RESULT_REQUEST_KEY,
                        result
                );
        dismiss();
    }
}

// =========================================================
// CATEGORY PICKER BOTTOM SHEET
// =========================================================
private void showCategoryPickerBottomSheet(
        UnitCategory currentCategory
) {
    if (currentCategory == null) {
        return;
    }
    /*
     * Tránh mở nhiều Category Picker chồng nhau.
     */
    if (getSupportFragmentManager()
            .findFragmentByTag(
                    TAG_CATEGORY_PICKER_BOTTOM_SHEET
            ) != null) {
        return;
    }
    CategoryPickerBottomSheet bottomSheet =
            CategoryPickerBottomSheet.newInstance(
                    currentCategory
            );
    bottomSheet.show(
            getSupportFragmentManager(),
            TAG_CATEGORY_PICKER_BOTTOM_SHEET
    );
}

private void handleCategoryPickerResult(
        Bundle resultBundle
) {
    if (resultBundle == null
            || converterController == null) {
        return;
    }
    String categoryName =
            resultBundle.getString(
                    CategoryPickerBottomSheet
                            .RESULT_CATEGORY
            );
    if (categoryName == null
            || categoryName.trim().isEmpty()) {
        return;
    }
    UnitCategory selectedCategory;
    try {
        selectedCategory =
                UnitCategory.valueOf(
                        categoryName
                );
    } catch (IllegalArgumentException exception) {
        return;
    }
    converterController.selectCategory(
            selectedCategory
    );
}
