// =========================================================

/**
 * Mở Calculation History dưới dạng Bottom Sheet.
 */
private void setupHistoryButton() {
    findViewById(
            R.id.btnHistory
    ).setOnClickListener(view -> {
        /*
         * Không mở thêm một HistoryBottomSheet mới
         * nếu sheet hiện tại đã tồn tại.
         *
         * Tránh trường hợp user tap liên tục:
         *
         * History
         * History
         * History
         *
         * → tạo nhiều Fragment chồng nhau.
         */
        if (getSupportFragmentManager()
                .findFragmentByTag(
                        TAG_HISTORY_BOTTOM_SHEET
                ) != null) {
            return;
        }
        HistoryBottomSheet historyBottomSheet =
                new HistoryBottomSheet();
        historyBottomSheet.show(
                getSupportFragmentManager(),
                TAG_HISTORY_BOTTOM_SHEET
        );
    });
}
