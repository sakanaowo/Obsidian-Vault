package com.example.calculator.converter;
import com.example.calculator.model.UnitCategory;
import com.example.calculator.model.UnitDefinition;
import com.example.calculator.utils.NumberFormatter;
/**
 * Xử lý business logic của Unit Converter.
 *
 * Engine không phụ thuộc Android UI.
 *
 * Với category thông thường:
 *
 * input
 *      ↓
 * input × from.factorToBase
 *      ↓
 * base value
 *      ↓
 * base value ÷ to.factorToBase
 *      ↓
 * result
 *
 * Temperature được xử lý riêng thông qua Kelvin:
 *
 * From temperature
 *      ↓
 * Kelvin
 *      ↓
 * To temperature
 */
public final class UnitConverterEngine {
    // =========================================================
    // TEMPERATURE SYMBOLS
    // =========================================================
    private static final String SYMBOL_CELSIUS =
            "°C";
    private static final String SYMBOL_FAHRENHEIT =
            "°F";
    private static final String SYMBOL_KELVIN =
            "K";
    // =========================================================
    // PUBLIC API
    // =========================================================
    /**
     * Chuyển đổi một giá trị giữa hai unit cùng category.
     *
     * Không throw exception ra UI.
     * Mọi lỗi được trả về bằng ConversionResult.
     */
    public ConversionResult convert(
            double inputValue,
            UnitDefinition fromUnit,
            UnitDefinition toUnit
    ) {
        // -----------------------------------------------------
        // INPUT VALIDATION
        // -----------------------------------------------------
        if (!isFinite(inputValue)) {
            return ConversionResult.error(
                    "Input value must be a finite number."
            );
        }
        if (fromUnit == null) {
            return ConversionResult.error(
                    "From unit cannot be null."
            );
        }
        if (toUnit == null) {
            return ConversionResult.error(
                    "To unit cannot be null."
            );
        }
        // -----------------------------------------------------
        // CATEGORY VALIDATION
        // -----------------------------------------------------
        if (fromUnit.getCategory()
                != toUnit.getCategory()) {
            return ConversionResult.error(
                    "Units must belong to the same category."
            );
        }
        // -----------------------------------------------------
        // SAME UNIT
        // -----------------------------------------------------
        /*
         * Ví dụ:
         *
         * 25 m → m
         * 30 °C → °C
         *
         * giữ nguyên input.
         */
        if (fromUnit.equals(toUnit)) {
            return ConversionResult.success(
                    inputValue
            );
        }
        // -----------------------------------------------------
        // TEMPERATURE
        // -----------------------------------------------------
        if (fromUnit.getCategory()
                == UnitCategory.TEMPERATURE) {
            return convertTemperature(
                    inputValue,
                    fromUnit,
                    toUnit
            );
        }
        // -----------------------------------------------------
        // STANDARD FACTOR CONVERSION
        // -----------------------------------------------------
        /*
         * Bước 1:
         *
         * From unit → base unit
         */
        double baseValue =
                inputValue
                        * fromUnit.getFactorToBase();
        if (!isFinite(baseValue)) {
            return ConversionResult.error(
                    "Conversion result is outside the supported range."
            );
        }
        /*
         * Bước 2:
         *
         * Base unit → To unit
         */
        double convertedValue =
                baseValue
                        / toUnit.getFactorToBase();
        if (!isFinite(convertedValue)) {
            return ConversionResult.error(
                    "Conversion result is outside the supported range."
            );
        }
        return ConversionResult.success(
                convertedValue
        );
    }
    // =========================================================
    // TEMPERATURE CONVERSION
    // =========================================================
    /**
     * Chuyển Temperature theo hai bước:
     *
     * fromUnit
     *      ↓
     * Kelvin
     *      ↓
     * toUnit
     */
    private ConversionResult convertTemperature(
            double inputValue,
            UnitDefinition fromUnit,
            UnitDefinition toUnit
    ) {
        Double kelvinValue =
                convertTemperatureToKelvin(
                        inputValue,
                        fromUnit
                );
        if (kelvinValue == null) {
            return ConversionResult.error(
                    "Unsupported source temperature unit."
            );
        }
        if (!isFinite(kelvinValue)) {
            return ConversionResult.error(
                    "Conversion result is outside the supported range."
            );
        }
        Double convertedValue =
                convertKelvinToTemperature(
                        kelvinValue,
                        toUnit
                );
        if (convertedValue == null) {
            return ConversionResult.error(
                    "Unsupported target temperature unit."
            );
        }
        if (!isFinite(convertedValue)) {
            return ConversionResult.error(
                    "Conversion result is outside the supported range."
            );
        }
        return ConversionResult.success(
                convertedValue
        );
    }
    /**
     * Chuyển Celsius / Fahrenheit / Kelvin
     * về Kelvin.
     */
    private Double convertTemperatureToKelvin(
            double value,
            UnitDefinition unit
    ) {
        String symbol =
                unit.getSymbol();
        if (SYMBOL_CELSIUS.equals(symbol)) {
            /*
             * K = °C + 273.15
             */
            return value
                    + 273.15;
        }
        if (SYMBOL_FAHRENHEIT.equals(symbol)) {
            /*
             * K = (°F - 32) × 5/9 + 273.15
             */
            return (value - 32.0)
                    * 5.0
                    / 9.0
                    + 273.15;
        }
        if (SYMBOL_KELVIN.equals(symbol)) {
            return value;
        }
        return null;
    }
    /**
     * Chuyển Kelvin sang Celsius / Fahrenheit / Kelvin.
     */
    private Double convertKelvinToTemperature(
            double kelvinValue,
            UnitDefinition unit
    ) {
        String symbol =
                unit.getSymbol();
        if (SYMBOL_CELSIUS.equals(symbol)) {
            /*
             * °C = K - 273.15
             */
            return kelvinValue
                    - 273.15;
        }
        if (SYMBOL_FAHRENHEIT.equals(symbol)) {
            /*
             * °F = (K - 273.15) × 9/5 + 32
             */
            return (kelvinValue - 273.15)
                    * 9.0
                    / 5.0
                    + 32.0;
        }
        if (SYMBOL_KELVIN.equals(symbol)) {
            return kelvinValue;
        }
        return null;
    }
    // =========================================================
    // VALIDATION
    // =========================================================
    private boolean isFinite(
            double value
    ) {
        return !Double.isNaN(value)
                && !Double.isInfinite(value);
    }
    // =========================================================
    // RESULT MODEL
    // =========================================================
    /**
     * Kết quả trả về cho UI Controller.
     *
     * UI không cần catch exception
     * và không tự format số.
     */
    public static final class ConversionResult {
        private final boolean success;
        private final double value;
        private final String formattedValue;
        private final String errorMessage;
        private ConversionResult(
                boolean success,
                double value,
                String formattedValue,
                String errorMessage
        ) {
            this.success =
                    success;
            this.value =
                    value;
            this.formattedValue =
                    formattedValue;
            this.errorMessage =
                    errorMessage;
        }
        private static ConversionResult success(
                double value
        ) {
            return new ConversionResult(
                    true,
                    value,
                    NumberFormatter.format(
                            value
                    ),
                    null
            );
        }
        private static ConversionResult error(
                String errorMessage
        ) {
            return new ConversionResult(
                    false,
                    Double.NaN,
                    "Error",
                    errorMessage
            );
        }
        public boolean isSuccess() {
            return success;
        }
        /**
         * Chỉ đọc khi isSuccess() == true.
         */
        public double getValue() {
            return value;
        }
        /**
         * Chuỗi đã được NumberFormatter xử lý.
         */
        public String getFormattedValue() {
            return formattedValue;
        }
        /**
         * null khi conversion thành công.
         */
        public String getErrorMessage() {
            return errorMessage;
        }
    }
}

Công thức đang sử dụng

Celsius → Kelvin

K = °C + 273.15

Fahrenheit → Kelvin

K = (°F − 32) × 5/9 + 273.15

Kelvin → Celsius

°C = K − 273.15

Kelvin → Fahrenheit

°F = (K − 273.15) × 9/5 + 32

Mọi phép đổi đều đi qua Kelvin:

°C → K → °F
°F → K → °C
K  → K → °C
