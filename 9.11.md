5. Thêm showUnitPickerBottomSheet()

Đặt dưới nhóm Converter Picker:

// =========================================================
// UNIT PICKER BOTTOM SHEET
// =========================================================
/**
 * Mở Unit Picker cho From hoặc To.
 */
private void showUnitPickerBottomSheet(
        UnitCategory category,
        UnitDefinition currentUnit,
        String target
) {
    if (category == null
            || currentUnit == null) {
        return;
    }
    /*
     * Tránh mở nhiều Unit Picker chồng nhau
     * khi user nhấn selector liên tục.
     */
    if (getSupportFragmentManager()
            .findFragmentByTag(
                    TAG_UNIT_PICKER_BOTTOM_SHEET
            ) != null) {
        return;
    }
    UnitPickerBottomSheet bottomSheet =
            UnitPickerBottomSheet.newInstance(
                    category,
                    currentUnit,
                    target
            );
    bottomSheet.show(
            getSupportFragmentManager(),
            TAG_UNIT_PICKER_BOTTOM_SHEET
    );
}

6. Thêm Fragment Result Listener

Thêm method:

/**
 * Nhận unit sau khi user bấm Confirm.
 */
private void setupUnitPickerResultListener() {
    getSupportFragmentManager()
            .setFragmentResultListener(
                    UnitPickerBottomSheet.RESULT_REQUEST_KEY,
                    this,
                    (requestKey, resultBundle) -> {
                        handleUnitPickerResult(
                                resultBundle
                        );
                    }
            );
}

7. Thêm handleUnitPickerResult()

/**
 * Đọc kết quả từ UnitPickerBottomSheet
 * rồi cập nhật From hoặc To unit.
 */
private void handleUnitPickerResult(
        Bundle resultBundle
) {
    if (resultBundle == null
            || converterController == null
            || unitRepository == null) {
        return;
    }
    String target =
            resultBundle.getString(
                    UnitPickerBottomSheet.RESULT_TARGET
            );
    String categoryName =
            resultBundle.getString(
                    UnitPickerBottomSheet.RESULT_CATEGORY
            );
    String unitSymbol =
            resultBundle.getString(
                    UnitPickerBottomSheet.RESULT_UNIT_SYMBOL
            );
    if (categoryName == null
            || unitSymbol == null
            || unitSymbol.trim().isEmpty()) {
        return;
    }
    UnitCategory selectedCategory;
    try {
        selectedCategory =
                UnitCategory.valueOf(
                        categoryName
                );
    } catch (IllegalArgumentException exception) {
        return;
    }
    /*
     * Không nhận kết quả cũ nếu category của Converter
     * đã thay đổi trong lúc Unit Picker đang mở.
     */
    if (selectedCategory
            != converterController.getSelectedCategory()) {
        return;
    }
    UnitDefinition selectedUnit =
            unitRepository.findBySymbol(
                    selectedCategory,
                    unitSymbol
            );
    if (selectedUnit == null) {
        return;
    }
    if (UnitPickerBottomSheet.TARGET_FROM.equals(
            target
    )) {
        converterController.selectFromUnit(
                selectedUnit
        );
        return;
    }
    if (UnitPickerBottomSheet.TARGET_TO.equals(
            target
    )) {
        converterController.selectToUnit(
                selectedUnit
        );
    }
}

Bundle đã được import trong MainActivity thông qua:

import android.os.Bundle;

nên không cần thêm import mới cho phần này.
